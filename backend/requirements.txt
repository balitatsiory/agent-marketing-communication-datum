# Dépendances de l'API IAGORA.
#
# FastAPI n'a jamais atteint la 1.0 et introduit des ruptures en version MINEURE :
# les 0.160.0 et 0.162.0 sont marquées rupturantes, et la 0.162.6 a retiré le
# support de Python 3.10. Un plancher ouvert (fastapi>=0.115) ramènerait donc du
# code cassé sans prévenir. La version est bornée, et on en change par décision.

# --- API ---
fastapi>=0.163,<0.164
uvicorn[standard]>=0.30,<1.0

# --- validation et configuration ---
pydantic>=2.7,<3.0
pydantic-settings>=2.3,<3.0
email-validator>=2.1           # validation des adresses (EmailStr)

# --- base de données ---
# Le suffixe [asyncio] est OBLIGATOIRE depuis SQLAlchemy 2.1 : greenlet, dont
# dépend tout le code asynchrone, ne s'installe plus par défaut.
sqlalchemy[asyncio]>=2.1,<2.2
asyncpg>=0.29,<1.0             # pilote PostgreSQL asynchrone
alembic>=1.13,<2.0

# --- sécurité ---
bcrypt>=4.1,<6.0               # empreinte des mots de passe
pyjwt>=2.8,<3.0                # jetons d'accès
cryptography>=42.0             # chiffrement des secrets de plateformes (Fernet)

# --- appels sortants ---
httpx>=0.27,<1.0

# --- envoi de courriels ---
# smtplib, de la bibliotheque standard, est BLOQUANT : un serveur lent y
# immobiliserait la boucle d'evenements, donc toute l'API. aiosmtplib parle le
# meme protocole en asynchrone, SSL/TLS implicite (465) et STARTTLS (587)
# compris.
aiosmtplib>=3.0,<5.0

# --- stockage d'objets ---
# Cloudflare R2 parle le protocole S3 : le client est celui d'AWS, pointé sur
# l'adresse R2 du compte. boto3 est synchrone, mais la signature d'URL et les
# appels de controle ne font pas d'entree-sortie bloquante longue ; le depot
# d'un fichier, lui, passe par un fil d'execution separe (run_in_threadpool),
# pour ne pas immobiliser la boucle evenementielle.
boto3>=1.35,<2.0

# --- traitement d'images ---
# Compression au-dela de 8 Mo, et lecture des dimensions au depot.
pillow>=10.4,<12.0

# --- formulaires multipart ---
# Necessaire a FastAPI pour lire un fichier depose (UploadFile).
python-multipart>=0.0.9,<1.0
